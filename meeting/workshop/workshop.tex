\section{木幅（treewidth）}

\begin{frame}{木幅（treewidth）とは}
    \begin{itemize}
        \setlength{\parskip}{0.8em}
        \item 定義：頂点集合の木構造を用いて定義される正の整数の値\cite{david2025treewidth}.
        \item 「グラフを木のように分解できる度合い」を示す.
        \item 値が小さいほど「木」に近い.
        \item グラフ構造理論などの中心的な概念（応用分野：ベイズ推論や制御理論）.
    \end{itemize}
    \begin{figure}
    \vspace{1em}
    \includegraphics[width=0.65\textwidth]{example_treewidth.png}
    \caption{木幅の例}
    \end{figure}
\end{frame}


\begin{frame}{木幅による NP 困難性の緩和}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item \textbf{巡回セールスマン問題（TSP）\cite{bodlaender2015deterministic}}
        \item 一般のグラフではNP困難
        \item 計算量
            \begin{itemize}
                \setlength{\itemsep}{0.8em}
                \item held-karpアルゴリズム：\(O(2^{n} n^{2})\)
                \item 現在知られている解法はいずれも指数時間を要する.
                \item 木幅 \(k\) である場合：\(2^{O(k)} n^{O(1)}\)
            \end{itemize}
      \end{itemize}

\begin{frame}{木幅の求め方}

    \textbf{木幅アルゴリズム}
    \vspace{1em}
    \begin{enumerate}

      \item 袋サイズの\textbf{最大値 − 1} が木幅.
    \end{enumerate}

\section{木分解(tree-decomposition)}
\begin{frame}{木分解(tree-decomposition)の3つの条件}
    以下の全ての条件を満たしているとき,木分解から木幅を求めることができる\cite{heinz2013tree}.
    \begin{columns}
        \begin{column}{0.58\textwidth}
           \begin{itemize}
             \item \textbf{条件1（頂点の網羅）}  \\
                グラフのすべての頂点は，どこかの袋に必ず入っている.
             \item \textbf{条件2（辺の保存）}  \\
                辺でつながっている2つの頂点は，同じ袋に一度は一緒に現れる.
             \item \textbf{条件3（連続性）}  \\
                ある頂点を含む袋は，木の上で途切れずにつながっている.
            \end{itemize} 
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{figure}
                \includegraphics[width=\textwidth]{tree-decomposition-rule.png}
                \caption{木分解の条件}
            \end{figure}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{定義3.1.1\cite{heinz2013tree}}
        グラフ $G=(V,E)$ の木分解 (tree-decomposition)とは，木 $T$ と，$T$ の各頂点 $t$ に対応する頂点集合 $V_t$ の族 $\mathcal{V}=(V_t)_{t\in V(T)}$ からなる組 $(T,\mathcal{V})$ であって，次の性質を満たすものである.
        \vspace{1em}
    \begin{enumerate}
        {\small
        \setlength{\parskip}{1em}
        \item[(TD1)] $\displaystyle\textstyle\bigcup_{t\in V(T)} V_t = V$.\ (すべての頂点は少なくとも1つのバッグに現れる.)
        \item[(TD2)] 任意の辺 $uv\in E$ に対して，ある $t\in V(T)$ が存在し，$u,v\in V_t$ となる.\\(各辺の両端点は少なくとも1つの同じバッグに含まれる.)
        \item[(TD3)] 任意の $t_1,t_2,t_3\in V(T)$ に対して，もし $t_2$ が $t_1$ と $t_3$ を結ぶ経路上にあるならば，
        \[ V_{t_1}\cap V_{t_3} \subseteq V_{t_2}. \]
        (ある頂点 $v\in V$ が複数のバッグに現れるとき，それらのバッグは $T$ 上で連結な部分木を形成する（"接続性条件"）．)
        }
    \end{enumerate}
\end{frame}

\begin{frame}{例}
    5頂点の以下のグラフで以下のような木構造を考える.
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{figure}
                \includegraphics[scale=0.3]{5_treewidth.png}
            \end{figure}
        \end{column}
        \begin{column}{0.2\textwidth}
            \begin{figure}
            \includegraphics[scale=0.3]{5_treewidth_2.png}
            \end{figure}
        \end{column}
        \begin{column}{0.28\textwidth}
            \begin{itemize}
                \item $X_1=\{v_1,v_2,v_5\}$
                \item $X_2=\{v_2,v_3,v_5\}$
                \item $X_3=\{v_3,v_4,v_5\}$
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{itemize}

        \item TD1(すべての頂点は少なくとも1つのバッグに現れる)\ ◎
        \item TD2(各辺の両端点は少なくとも1つの同じバッグに含まれる.)\ ◎
        \item TD3(ある頂点 $v\in V$ が複数のバッグに現れるとき，それらのバッグは $T$ 上で連結な部分木を形成する．)\ ◎
        \item よって,TD1,TD2,TD3をすべて満たしているため木幅は3-1=2となる.
    \end{itemize}
    
\end{frame}
